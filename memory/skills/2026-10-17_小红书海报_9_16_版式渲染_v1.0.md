## 复用代码 (核心 Python pipeline)

```python
"""
v3.2 范式 9:16 渲染 pipeline — 2026-10-17 沉淀
"""
import os, shutil
from playwright.sync_api import sync_playwright

SRC = r"D:\B166ER-openclaw\workspace\memory\小红书海报"
DST = r"C:\Users\ericz\Desktop\小红书海报_酒店选址_2026-10-17"

THEMES = {
    "_xhs_sakura": {
        "#1a0e08": "#1a0a14", "#fff0e0": "#fff0f4",
        "#e07020": "#e8a0b8", "#a07040": "#a07080",
        "#241408": "#1a1424", "#1f1208": "#1a0c14",
        "#e87060": "#d8a0b0", "#80c870": "#a890c0",
        "rgba(224,112,32,": "rgba(232,160,184,",
        "rgba(180,80,80,": "rgba(216,160,180,",
        "rgba(80,180,120,": "rgba(160,144,192,",
    },
    "_luxe_gold": {
        "#1a0e08": "#0a0a0f", "#fff0e0": "#e8dcc8",
        "#e07020": "#d4a74a", "#a07040": "#6a5a4a",
        "#241408": "#1a1410", "#1f1208": "#0d0a06",
        "#e87060": "#b89060", "#80c870": "#a0b070",
        "rgba(224,112,32,": "rgba(212,167,74,",
        "rgba(180,80,80,": "rgba(216,180,120,",
        "rgba(80,180,120,": "rgba(160,160,112,",
    },
}

def render_template(template_basename, output_ratio="9x16", width=1080, height=1920, device_scale=2):
    base = open(os.path.join(SRC, f"{template_basename}.html"), encoding="utf-8").read()
    
    # 1. 生成色变体
    for suffix, mapping in THEMES.items():
        txt = base
        for old, new in mapping.items():
            txt = txt.replace(old, new)
        out_html = os.path.join(SRC, f"{template_basename}{suffix}.html")
        open(out_html, "w", encoding="utf-8").write(txt)
    
    # 2. 渲染
    files = [f"{template_basename}.html", f"{template_basename}_xhs_sakura.html", f"{template_basename}_luxe_gold.html"]
    with sync_playwright() as p:
        browser = p.chromium.launch(args=["--no-sandbox", "--disable-dev-shm-usage"])
        for html in files:
            path = os.path.join(SRC, html)
            png = path.replace(".html", ".png")
            with open(path, encoding="utf-8") as f: content = f.read()
            page = browser.new_page(viewport={"width": width, "height": height}, device_scale_factor=device_scale)
            page.set_content(content, wait_until="domcontentloaded")
            page.wait_for_timeout(500)
            page.screenshot(path=png, full_page=False, omit_background=False, timeout=15000)
            page.close()
        browser.close()
    
    # 3. 同步桌面
    os.makedirs(DST, exist_ok=True)
    for f in os.listdir(SRC):
        if f.endswith(".png") and template_basename in f:
            shutil.copy(os.path.join(SRC, f), DST)
    print(f"DONE: 3 PNG synced to {DST}")

# 调用示例
render_template("hotel_location_v3_2_9x16_brand_orange", output_ratio="9x16")
```
