api_key = ***"TAVILY_API_KEY", "")
assert len(api_key) == 57, "TAVILY_API_KEY 长度异常"

# === Step 2: Tavily search (AI 摘要) ===
from tavily import TavilyClient
client = TavilyClient(api_key=***
def tavily_search(query, max_results=3):
    r = client.search(query=query, max_results=max_results)
    return r["results"]  # [{"title": ..., "url": ..., "content": ...}]

# === Step 3: Scrapling 0.5 fetch (隐身爬) ===
from scrapling import StealthyFetcher, Fetcher

def scrapling_fetch(url, stealth=False, headless=True, timeout_ms=20000, retries=3):
    """⚠️ timeout 是毫秒! 20s = 20000"""
    if stealth:
        return StealthyFetcher.fetch(
            url, headless=headless, network_idle=True,
            timeout=timeout_ms, retries=retries,
            solve_cloudflare=False,  # 自动绕过 Cloudflare
        )
    else:
        return Fetcher.get(url, timeout=timeout_ms, retries=retries)

def scrapling_to_markdown(page) -> str:
    status = getattr(page, "status", "?")
    text = page.get_text() if hasattr(page, "get_text") else str(page)[:500]
    return f"<!-- status: {status} -->\n\n{text}"

# === Step 4: e2e (Tavily 找 → Scrapling 抓 → markdown 输出) ===
def e2e(query, output_path=None):
    md = [f"# {query} 抓取结果\n"]
    md.append(f"> 时间: {Path(__file__).stat().st_mtime}")
    md.append(f"> 工具: Tavily 0.8.5 + Scrapling 0.5\n")
    
    # Tavily
    results = tavily_search(query, max_results=3)
    md.append("## Tavily 摘要\n")
    for i, item in enumerate(results, 1):
        md.append(f"{i}. **{item['title'][:80]}**")
        md.append(f"   URL: {item['url'][:80]}")
        md.append(f"   snippet: {item.get('content', '')[:100]}\n")
    
    # Scrapling 抓 Tavily 找到的 URL
    if results:
        url = results[0]["url"]
        md.append(f"\n## Scrapling 抓取 {url[:60]}\n")
        try:
            page = scrapling_fetch(url, stealth=True)
            md.append(scrapling_to_markdown(page))
        except Exception as e:
            md.append(f"❌ Scrapling 失败: {e}")
    
    text = "\n".join(md)
    if output_path:
        Path(output_path).write_text(text, encoding="utf-8")
        print(f"[OK] saved: {output_path}")
    return text

# 调用示例

## 已知限制

1. **Tavily 配额**: API key 有 quota, 需监控 (历史 quota 400 错误)
2. **Scrapling timeout**: 毫秒单位 (易错, 默认 20ms = 立即超时)
3. **Scrapling 0.5 API**: `from scrapling import Fetcher/StealthyFetcher` (不是 `scrapling.fetch`)
4. **mct.gov.cn 等政府站**: 部分 URL 返 404 (网站改版), 需重定向 URL
5. **隐身爬虫**: solve_cloudflare=True 慢 (10-30s), 普通站 False 即可
6. **.env 文件**: 必须 ASCII 编码 (无 UTF-8 BOM), 否则 Python 看不到
